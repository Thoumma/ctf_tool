auto_all.sh — master dispatcher (keyboard input)

Type any target at the prompt (or pass as arg). Detects URL vs pcap vs memory dump vs media/binary/archive vs folder, runs the right checker (capturing its output), then builds the aggregated report.md. A directory routes to Hayabusa only if it contains .evtx, else to auto_dirsweep.sh. Falls back to file-content sniffing when the extension lies.

#!/usr/bin/env bash
# auto_all.sh [target] — type the target at the prompt, or pass it as an arg.
set -u
HERE="$(cd "$(dirname "$(readlink -f "$0")")" && pwd)"
if [ -n "${1:-}" ]; then T="$1"; else
  read -rp "[?] target (file.pcap / dump.mem / http://host): " T
fi
[ -n "$T" ] || { echo "nothing given"; exit 1; }

if printf '%s' "$T" | grep -qiE '^https?://'; then
  echo "[*] URL -> auto_web.py"; exec "$HERE/auto_web.py" "$T"
fi
[ -d "$T" ] && exec "$HERE/auto_hayabusa.sh" "$T"   # directory of evtx -> Hayabusa
[ -f "$T" ] || { echo "[-] not a URL and not a file: $T"; exit 1; }
ext="${T##*.}"; ext="${ext,,}"; desc="$(file -b "$T")"; echo "[*] file: $desc"

case "$ext" in
  pcap|pcapng|cap)                      exec "$HERE/auto_pcap.sh" "$T";;
  mem|raw|vmem|dmp|lime|vmss|vmsn|dump|core) exec "$HERE/auto_mem.sh" "$T";;
  evtx)                                 exec "$HERE/auto_hayabusa.sh" "$T";;  # or auto_winlog.py (no binary)
  png|jpg|jpeg|bmp|gif|wav|mp3|flac|ogg|svg|pdf|docx|doc|xlsx|xls|pptx|ppt)
                                        exec "$HERE/auto_forensic.sh" "$T";;
  txt|enc|b64|hex|cipher)               exec "$HERE/auto_crypto.py" "$(cat "$T")";;
  zip|rar|7z|tar|gz) echo "[*] archive -> 7z l '$T'; zip2john/bkcrack if locked"; exit 0;;
esac
case "$desc" in                          # extension unknown -> sniff content
  *pcap*|*"capture file"*)              exec "$HERE/auto_pcap.sh" "$T";;
  *image*|*PNG*|*JPEG*|*bitmap*|*GIF*)  exec "$HERE/auto_forensic.sh" "$T";;
  *ELF*|*PE32*|*executable*|*Mach-O*)   echo "[*] binary -> rev/pwn"; command -v checksec >/dev/null && checksec --file="$T"; exit 0;;
  *ASCII*|*text*|*Unicode*)             exec "$HERE/auto_crypto.py" "$(cat "$T")";;
  *data*) echo "[?] raw data — guessing memory"; exec "$HERE/auto_mem.sh" "$T";;
  *) echo "[?] unknown — triage"; exec "$HERE/auto_triage.sh" "$T";;
esac

Identifies the file, greps for the flag (raw + base64-decoded), then prints the exact next command for its type.

#!/usr/bin/env bash
# auto_triage.sh <file>  —  fingerprint anything and route to the right deep-check.
set -u
if [ -n "${1:-}" ]; then F="$1"; else read -rp "[?] file: " F; fi
[ -n "$F" ] || { echo "no file given"; exit 1; }
have(){ command -v "$1" >/dev/null 2>&1; }
line(){ printf '\n\033[1;36m=== %s ===\033[0m\n' "$*"; }

line "file type";   file "$F"
line "size";        printf '%s bytes\n' "$(wc -c < "$F")"
line "magic (hex)"; xxd "$F" | head -3
have ent && { line "entropy (hi≈encrypted)"; ent "$F" | grep -i entropy; }

line "FLAG grep (raw)"
grep -aoE '[A-Za-z0-9_]{2,}\{[^}]{1,120}\}' "$F" | sort -u
line "FLAG grep (base64-decoded chunks)"
grep -aoE '[A-Za-z0-9+/=]{16,}' "$F" | while read -r s; do echo "$s" | base64 -d 2>/dev/null; done \
  | grep -aoE '[A-Za-z0-9_]{2,}\{[^}]{1,120}\}' | sort -u

line "strings hits"
{ strings -n 6 "$F"; strings -e l "$F"; } | grep -aiE 'flag|ctf|secret|passw|key' | head
line "embedded files"; have binwalk && binwalk "$F"

line "ROUTE"
case "$(file -b "$F")" in
  *PNG*|*JPEG*|*image*|*bitmap*|*GIF*)   echo "-> ./auto_forensic.sh '$F'   (media/stego)";;
  *filesystem*|*"disk image"*|*"boot sector"*) echo "-> mmls '$F'; fls -r -o <off> '$F'; autopsy";;
  *pcap*|*"capture file"*|*tcpdump*)     echo "-> ./auto_pcap.sh '$F'";;
  *ELF*|*PE32*|*executable*|*Mach-O*)    echo "-> checksec --file='$F'; ltrace './$F'; ghidra";;
  *Zip*|*RAR*|*7-zip*|*archive*|*gzip*)  echo "-> 7z l '$F'   (locked? zip2john/bkcrack)";;
  *PDF*)                                 echo "-> pdfid '$F'; pdftotext -layout '$F' -; olevba";;
  *ASCII*|*text*|*Unicode*)              echo "-> ./auto_crypto.py \"\$(cat '$F')\"";;
  *)                                     echo "-> binwalk -e '$F'; foremost -i '$F' -o carved/";;
esac

#!/usr/bin/env bash
# auto_pcap.sh <capture.pcap>  —  full PCAP triage + flag sweep.
set -u
if [ -n "${1:-}" ]; then SRC="$1"; else read -rp "[?] pcap: " SRC; fi
F="$(readlink -f "$SRC")"
OUT="pcap_$(basename "$F")_out"; mkdir -p "$OUT"; cd "$OUT" || exit 1
run(){ printf '\n\033[1;36m=== %s ===\033[0m\n' "$*"; "$@" 2>&1 | head -60; }

command -v capinfos >/dev/null && run capinfos "$F"
run tshark -r "$F" -q -z io,phs
run tshark -r "$F" -q -z conv,tcp

for proto in http smb tftp imf; do
  mkdir -p "obj_$proto"; tshark -r "$F" --export-objects "$proto,obj_$proto" >/dev/null 2>&1
done
run tshark -r "$F" -Y 'http.request' -T fields -e http.host -e http.request.uri
tshark -r "$F" -Y 'dns' -T fields -e dns.qry.name 2>/dev/null | sort -u > dns_names.txt

echo "[*] cleartext creds (FTP/HTTP-basic):"
tshark -r "$F" -Y 'ftp.request.command=="USER" || ftp.request.command=="PASS"' -T fields -e ftp.request.arg 2>/dev/null
tshark -r "$F" -Y 'http.authorization' -T fields -e http.authorization 2>/dev/null

echo "[*] USB HID -> usb_hid.txt (feed to ctf-usb-keyboard-parser):"
tshark -r "$F" -Y 'usb.capdata' -T fields -e usb.capdata 2>/dev/null | sed 's/://g' > usb_hid.txt

command -v tcpflow  >/dev/null && { mkdir -p streams; tcpflow -r "$F" -o streams >/dev/null 2>&1; }
command -v foremost >/dev/null && foremost -i "$F" -o carved >/dev/null 2>&1

printf '\n\033[1;32m=== FLAG SWEEP ===\033[0m\n'

def printable(b):
    t = b.decode("latin-1","ignore")
    return bool(t) and sum(c in string.printable for c in t)/len(t) > 0.9

def decodes(data):
    s = re.sub(r"\s","", data.decode("latin-1","ignore")); out = {}
    for n,fn in (("b64",lambda x:base64.b64decode(x+"===")),("b32",lambda x:base64.b32decode(x+"======")),
                 ("b16",lambda x:base64.b16decode(x.upper())),("b85",lambda x:base64.b85decode(x))):
        try:
            d=fn(s)
            if d and printable(d): out[n]=d
        except Exception: pass
    try:
        if re.fullmatch(r"[0-9a-fA-F]+",s): out["hex"]=bytes.fromhex(s)
    except Exception: pass

    out["rot13"]=codecs.encode(data.decode("latin-1","ignore"),"rot_13").encode()

        x=bytes(b^k for b in data)
        if FLAG.search(x.decode("latin-1","ignore")): out[f"xor{k:#x}"]=x; break
    return out

def recurse(data, d=0, seen=None):
    seen=seen or set()
    for m in FLAG.findall(data.decode("latin-1","ignore")): print("  "*d+f"[FLAG] {m}")
    if d>=6 or data in seen: return

def rc4(key, data):
    S = list(range(256)); j = 0
    for i in range(256):
        j = (j + S[i] + key[i % len(key)]) % 256; S[i], S[j] = S[j], S[i]
    i = j = 0; out = bytearray()
    for ch in data:
        i = (i+1) % 256; j = (j+S[i]) % 256; S[i], S[j] = S[j], S[i]
        out.append(ch ^ S[(S[i]+S[j]) % 256])
    return bytes(out)

⭐ The recurring 2026 meta-patterns (read first)

parser-differentialattack-the-substrateinstrument-dont-reimplement
1. Parser / validator differential. Two components read the same bytes differently — validator vs executor, auditor vs runtime, Python vs native, gateway vs worker, signature-len vs parse-len, LE vs BE, single-decode vs double-decode. The gap is the bug. (z0d1ak Middle-Out/Control-Plane, ASIS 2048, D3 Scope-Drift, L3ak Zebda/ml-platform, BDSec Muktir, Anti-Slop Slipstream, GPN stupidcontract, OmniCTF CredVault.)
2. Attack the substrate, not the primitive. The crypto core is sound; the wrapper leaks it — a hint field, public storage slot, reused nonce/keystream, printed-not-evaluated value, malleable ciphertext. (z0d1ak crypto set, UIUCTF crypto, ASIS Hackel.)
3. Instrument the original binary; don't reimplement. Run the challenge's own verifier under GDB / Unicorn / Wine / code-cave hooks / msolve rather than cloning its custom crypto. (UIUCTF RE, ASIS LeakMeAk, KaliTeam Whispering-Feather, L3ak Yet-Another-Chat.)
4. Every obfuscation layer is a bijective codec — invert from the target backward. ARX/Feistel/VM chains, self-inverse transforms, reversible automata. (z0d1ak stars-below, BushBash ⟨⟩×6, Subleq-Scramble, autocooker.)
5. The handout / name is the oracle. Ship-code = game logic, operator names, template, corpus. And strings | grep flag / reading every switch case still wins (Midnight-Sun riscal, OmniCTF, BYU). The flag name often telegraphs the attack class in leetspeak.
